# BambouAI — complete route inventory (bambouaibot.com)
# Recovered 2026-10-18 from the official contract (/openapi.json, v2026-10-04)
# + the dashboard bundle (/app.js, md5 617d54cc0548ffb806f87e49036ca5e2).
# [OPENAPI] = documented in the public contract.  [BUNDLE] = internal supervision
# route used by the dashboard, same session cookie, not in the public contract.
#
# Auth modes:
#   session  = studio session cookie "bambou_session" (POST /api/login)
#   ext      = X-Bambou-Extension-Key header
#   bbk      = Authorization: Bearer bbk_... (or X-Bambou-Assistant-Key)
#   -        = no auth

GET     /health                                          [-]        [OPENAPI]  service probe -> {ok, service:"bambou-dashboard"}
POST    /api/login                                       [-]        [OPENAPI]  {"key":"<login key>"} -> Set-Cookie bambou_session (12h)
GET     /api/session                                     [session]  [OPENAPI]  {authenticated: bool}
POST    /api/logout                                      [session]  [OPENAPI]  drop session

# --- Assistants (assistant key bbk_...) -----------------------------------------
